### 3. ما أبقيتُه للشك (ب)

        if n['type'] == 'heading':

            continue
        if n['type'] != 'paragraph':
            chk(n == o, f'S{d}.{i}: كتلة {n["type"]} متطابقة'); continue
        rest = lambda p: {k: v for k, v in p.items() if k not in ('title', 'segments')}
        chk(rest(n) == rest(o), f'S{d}.{i}: role وkind وayahs وغيرها متطابقة')
        if n.get('title') != o.get('title'):
            changed.append((f'S{d}.{i} عنوان', o['title'], n['title'])); titleok(n['title'], o['title'], f'S{d}.{i}')
        chk([s['t'] for s in n['segments']] == [s['t'] for s in o['segments']], f'S{d}.{i}: تسلسل أنواع القطع')
        for k, (a, b) in enumerate(zip(n['segments'], o['segments'])):
            if a['t'] == 'text' and a != b: changed.append((f'S{d}.{i}[{k}]', b['v'], a['v']))
            if a['t'] != 'text': chk(a == b, f'S{d}.{i}[{k}]: قطعة {a["t"]} متطابقة')

        for s in n['segments']:
            for r in rid(s): chk(r in exp['records'] and exp['records'][r]['depth_min'] <= d, f'S{d}.{i}: السجل {r} موجود وdepth_min لا يزيد على {d}')
            if s['t'] == 'quote':

                chk(any(s['v'] in h for h in [rec['claim']] + [e['quote'] for e in rec['evidence']]), f'S{d}.{i}: اقتباس {s["record"]} جزء من دليله')
norm = lambda t: re.sub(r'[ً-ٰٟۖ-ۭـ]', '', t).translate(str.maketrans('أإآةى', 'اااهي'))
words = lambda t: re.findall(r'[ء-ي]+', norm(t))
triples = {tuple(w[i:i + 3]) for a in exp['ayahs'] for w in [words(a['text'])] for i in range(len(w) - 2)}
def qhits(doc):                                                          # ثلاث كلمات متتالية من آية داخل نص النظام أو عناوينه

             ([b.get('title') or ''] + [s['v'] for s in b.get('segments', []) if s['t'] == 'text'] + [b.get('text') or ''])]
    return [p for t in texts for w in [words(t)] for p in zip(w, w[1:], w[2:]) if p in triples]
nh, oh = qhits(new), qhits(old)

print('\nالنتيجة:', 'نجح' if not fails else f'فشل ({len(fails)})')

## ما بقي بلا حسم ولماذا

- **الصفوف 1 إلى 4 (`80-r013` و`80-r021`):** هل عبارة المختصر «منشغل بأكابر المشركين أملًا في هدايتهم» خبر سيرة يدخل في ق-080، أم شرح لمعنى الآيات يحمله الطبري أيضًا («رجاءَ أن يُسلِمَ»، `80-r105`)؟ عددتُها حرجًا بنص القرار لثلاثة أسباب:

  والحسم للمنسّق أو لصاحب الفكرة.
- **الصفوف 6 و7 و11 و12:** لا تُحسم إلا بنص الكتاب، وهو خارج السحابة (ق-131).

## أقوى اعتراض على هذه النتيجة

**(ب)**
- **على الحرج في الصفوف 1 إلى 3:** «المختصر» تفسير يشرح معنى الآيات. والآيات 5 إلى 10 نفسها تثبت أن النبي ﷺ أقبل على من استغنى وأعرض عن الأعمى. والآية 7 («وما عليك ألا يزكى») يُفهم منها أنه كان يرجو هدايته. فإن عُدّت عبارة المختصر معنى لا خبرًا، نقص الحرج ثلاث جمل، وصار الحكم «تُنشر بعد علاج محلي (4 مواضع)». أبقيتُ الحرج لأن «أكابر المشركين» تفصيل لا تقوله الآيات، والقرار يمنع وروده في اللمحة بلا نسبة. والتخفيف هنا هو «الحكم المتفائل» الذي حذّرت منه مراجعة النصر.
- **على الحذف في الصفوف 8 و9 و13:** حذفتُ ما أظنه صحيحًا في الكتاب. فابن كثير يورد الحديثين في هذين الموضعين على ما يبدو، والدعاء آخر كلام السعدي في السورة. لكن لا سجل يحمل ذلك، والمعيار هو السجل.
- **على المراجعة كلها:** هي قراءة من عائلة النموذج نفسها التي بنت النص، بلا فتح الكتاب ولا تشغيل المصدِّر. ومعظم جمل الفهم والتدبر اقتباسات حرفية تحمل نفسها، فقلة الملاحظات فيهما ترجع إلى طبيعة النص أكثر مما ترجع إلى قوة المراجعة.
- **على التبسيط:** ستة مواضع لفظية قليلة بالنسبة إلى «كلمات مألوفة». والسبب أن أغلب النص لفظ «المختصر» و«معاني الكلمات» والاقتباسات، وأني أبقيتُ كل ما فيه شك.
