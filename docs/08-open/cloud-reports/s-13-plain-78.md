# يقارن content/nasij/<رقم>.json الحالي بنسخته في الالتزام الأساس، ويفحص حمل الاقتباسات مقابل content/export.
import json, re, subprocess, sys
S, BASE = sys.argv[1], sys.argv[2]
raw = open(f'content/nasij/{S}.json', encoding='utf-8').read()
new = json.loads(raw)                                                    # JSON صالح وإلا يرمي خطأ
old_raw = subprocess.check_output(['git', 'show', f'{BASE}:content/nasij/{S}.json']).decode()
old = json.loads(old_raw)
exp = json.load(open(f'content/export/surah-{S}.json', encoding='utf-8'))
fails, changed = [], []
def chk(ok, msg):
    print(('PASS ' if ok else 'FAIL ') + msg)
    if not ok: fails.append(msg)
def titleok(n, o, tag):                                                  # العنوان: علامة الاستفهام وطوله
    chk(n.endswith('؟') == o.endswith('؟') and len(n.split()) <= max(8, len(o.split())), f'{tag}: ؟ وطول العنوان')
rid = lambda s: s.get('records') or ([s['record']] if 'record' in s else [])
chk(json.dumps(old, ensure_ascii=False, indent=2) + '\n' != old_raw or raw == json.dumps(new, ensure_ascii=False, indent=2) + '\n', 'تنسيق الملف (indent=2) كما كان')
chk(list(new) == list(old) and all(new[k] == old[k] for k in old if k != 'levels'), 'surah وstyle وheld وpassages متطابقة')
chk([l['depth'] for l in new['levels']] == [0, 1, 2, 3] and new['levels'][3] == old['levels'][3], 'أربع درجات بالترتيب، والدرجة 3 متطابقة')
for d in (0, 1, 2):
    nb, ob = new['levels'][d]['blocks'], old['levels'][d]['blocks']
    chk([b['type'] for b in nb] == [b['type'] for b in ob], f'الدرجة {d}: عدد الكتل وأنواعها ({len(nb)})')
    for i, (n, o) in enumerate(zip(nb, ob)):
        if n['type'] == 'heading':
            chk({k: v for k, v in n.items() if k != 'text'} == {k: v for k, v in o.items() if k != 'text'}, f'S{d}.{i}: عنوان قسم: غير النص متطابق')
            if n['text'] != o['text']: changed.append((f'S{d}.{i} عنوان قسم', o['text'], n['text'])); titleok(n['text'], o['text'], f'S{d}.{i}')
            continue
        if n['type'] != 'paragraph':
            chk(n == o, f'S{d}.{i}: كتلة {n["type"]} متطابقة'); continue
        rest = lambda p: {k: v for k, v in p.items() if k not in ('title', 'segments')}
        chk(rest(n) == rest(o), f'S{d}.{i}: role وkind وayahs وغيرها متطابقة')
        if n.get('title') != o.get('title'):
            changed.append((f'S{d}.{i} عنوان', o['title'], n['title'])); titleok(n['title'], o['title'], f'S{d}.{i}')
        chk([s['t'] for s in n['segments']] == [s['t'] for s in o['segments']], f'S{d}.{i}: تسلسل أنواع القطع')
        for k, (a, b) in enumerate(zip(n['segments'], o['segments'])):
            if a['t'] == 'text' and a != b: changed.append((f'S{d}.{i}[{k}]', b['v'], a['v']))
            if a['t'] != 'text': chk(a == b, f'S{d}.{i}[{k}]: قطعة {a["t"]} متطابقة')
        chk([r for s in n['segments'] for r in rid(s)] == [r for s in o['segments'] for r in rid(s)], f'S{d}.{i}: تسلسل معرّفات السجلات')
        for s in n['segments']:
            for r in rid(s): chk(r in exp['records'] and exp['records'][r]['depth_min'] <= d, f'S{d}.{i}: السجل {r} موجود وdepth_min لا يزيد على {d}')
            if s['t'] == 'quote':
                rec = exp['records'][s['record']]
                chk(any(s['v'] in h for h in [rec['claim']] + [e['quote'] for e in rec['evidence']]), f'S{d}.{i}: اقتباس {s["record"]} جزء من دليله')
norm = lambda t: re.sub(r'[ً-ٰٟۖ-ۭـ]', '', t).translate(str.maketrans('أإآةى', 'اااهي'))
words = lambda t: re.findall(r'[ء-ي]+', norm(t))
triples = {tuple(w[i:i + 3]) for a in exp['ayahs'] for w in [words(a['text'])] for i in range(len(w) - 2)}
def qhits(doc):                                                          # ثلاث كلمات متتالية من آية داخل نص النظام أو عناوينه
    texts = [t for d in (0, 1, 2) for b in doc['levels'][d]['blocks'] for t in
             ([b.get('title') or ''] + [s['v'] for s in b.get('segments', []) if s['t'] == 'text'] + [b.get('text') or ''])]
    return [p for t in texts for w in [words(t)] for p in zip(w, w[1:], w[2:]) if p in triples]
nh, oh = qhits(new), qhits(old)
chk(not [h for h in nh if nh.count(h) > oh.count(h)], 'لا ثلاث كلمات قرآنية متتالية جديدة في نص النظام والعناوين')
if oh: print('تنبيه: ثلاثيات قرآنية في نص الأساس نفسه (قبل أي تعديل):', oh)
print(f'\nالمواضع المتغيرة: {len(changed)}')
for c in changed: print(' ', c[0], '|', c[1], '=>', c[2])
print('\nالنتيجة:', 'نجح' if not fails else f'فشل ({len(fails)})')
sys.exit(1 if fails else 0)
