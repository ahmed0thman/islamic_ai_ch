            continue
        if n['type'] != 'paragraph':
            chk(n == o, f'S{d}.{i}: كتلة {n["type"]} متطابقة'); continue
        rest = lambda p: {k: v for k, v in p.items() if k not in ('title', 'segments')}
        chk(rest(n) == rest(o), f'S{d}.{i}: role وkind وayahs وغيرها متطابقة')
        if n.get('title') != o.get('title'):

            if a['t'] != 'text': chk(a == b, f'S{d}.{i}[{k}]: قطعة {a["t"]} متطابقة')

            if s['t'] == 'quote':

norm = lambda t: re.sub(r'[ً-ٰٟۖ-ۭـ]', '', t).translate(str.maketrans('أإآةى', 'اااهي'))
words = lambda t: re.findall(r'[ء-ي]+', norm(t))
triples = {tuple(w[i:i + 3]) for a in exp['ayahs'] for w in [words(a['text'])] for i in range(len(w) - 2)}
def qhits(doc):                                                          # ثلاث كلمات متتالية من آية داخل نص النظام أو عناوينه
    texts = [t for d in (0, 1, 2) for b in doc['levels'][d]['blocks'] for t in
             ([b.get('title') or ''] + [s['v'] for s in b.get('segments', []) if s['t'] == 'text'] + [b.get('text') or ''])]
    return [p for t in texts for w in [words(t)] for p in zip(w, w[1:], w[2:]) if p in triples]
nh, oh = qhits(new), qhits(old)

for c in changed: print(' ', c[0], '|', c[1], '=>', c[2])
print('\nالنتيجة:', 'نجح' if not fails else f'فشل ({len(fails)})')
sys.exit(1 if fails else 0)

**(ب)** أغلب جمل الفهم والتدبر اقتباسات حرفية تحمل نفسها بنفسها. فنسبة «محمولة» العالية (169 من 177) تعكس بناء النص أكثر مما تعكس قوة المراجعة.

والوجه الآخر لهذا البناء أخطر على هدف ق-087: **الألفاظ الصعبة في الفهم والتدبر داخل الاقتباسات أساسًا** («الضنك»، «الترافة والدعة»، «الاغترار»، «اعتسافه»، «نفائس الأنفاس»، «فحول الرجال»). والتبسيط لا يصل إليها لأن الاقتباس لا يُمس. فلغة الفهم والتدبر في هذه السورة بعد التبسيط ما زالت أصعب من مستوى «الإعدادية». **وقستُ ذلك (ب):** حروف الاقتباس الحرفي من حروف نص الفقرات (غير فقرات الرواية) 74.6% في الفهم (4277 من 5732)، و82.8% في التدبر (4340 من 5241)، وصفر في اللمحة. والحصص في `docs/02-method/nasij-reader-layer.md` («قواعد الصياغة» البند 4) 5% للفهم و20% للتدبر على الأكثر. وعلاج ذلك إعادة نسج تصوغ المعنى بدل اقتباسه (قاعدة الصياغة 2: «المعنى الإجمالي ومعنى الكلمة لا يُقتبسان: يُصاغان»)، وهو خارج حدود هذه المهمة التي تمنع الزيادة وتمنع مسّ الاقتباسات.

واعتراض ثانٍ: الحكم «تُنشر بعد علاج محلي» قد يُرى متساهلًا مع حرج قائم (`83-r012`). فإن عدّه المنسّق مانعًا قبل علاجه، فالحكم يصير «لا تُنشر حتى يُعالج `83-r012`». والعلاج في الحالتين سجلٌ واحد.
